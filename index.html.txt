// --- DRAGGABLE BUTTONS LOGIC ---
function makeDraggable(el) {
  var p1 = 0, p2 = 0, p3 = 0, p4 = 0;
  el.onmousedown = function(e) {
    e.preventDefault(); p3 = e.clientX; p4 = e.clientY;
    document.onmouseup = function() { document.onmouseup = null; document.onmousemove = null; };
    document.onmousemove = function(e) {
      e.preventDefault(); p1 = p3 - e.clientX; p2 = p4 - e.clientY; p3 = e.clientX; p4 = e.clientY;
      el.style.top = (el.offsetTop - p2) + "px"; el.style.left = (el.offsetLeft - p1) + "px";
    };
  };
  el.ontouchstart = function(e) {
    var t = e.touches[0]; p3 = t.clientX; p4 = t.clientY;
    document.ontouchend = function() { document.ontouchend = null; document.ontouchmove = null; };
    document.ontouchmove = function(e) {
      var t = e.touches[0]; p1 = p3 - t.clientX; p2 = p4 - t.clientY; p3 = t.clientX; p4 = t.clientY;
      el.style.top = (el.offsetTop - p2) + "px"; el.style.left = (el.offsetLeft - p1) + "px";
    };
  };
}

// --- GSTR-1 DATE FILTERS & RESTORED SUMMARY ---
function setGstrDates(range) {
  var today = new Date(); var y = today.getFullYear(); var m = today.getMonth();
  var fDate = "", tDate = "";
  if(range === 'this_month') {
      fDate = y + "-" + String(m+1).padStart(2,'0') + "-01";
      var lastDay = new Date(y, m+1, 0); tDate = y + "-" + String(m+1).padStart(2,'0') + "-" + String(lastDay.getDate()).padStart(2,'0');
  } else if(range === 'last_month') {
      var prevM = m === 0 ? 11 : m-1; var prevY = m === 0 ? y-1 : y;
      fDate = prevY + "-" + String(prevM+1).padStart(2,'0') + "-01";
      var lastDay = new Date(prevY, prevM+1, 0); tDate = prevY + "-" + String(prevM+1).padStart(2,'0') + "-" + String(lastDay.getDate()).padStart(2,'0');
  } else if(range === 'quarter') {
      var qStart = Math.floor(m/3)*3;
      fDate = y + "-" + String(qStart+1).padStart(2,'0') + "-01";
      var lastDay = new Date(y, qStart+3, 0); tDate = y + "-" + String(qStart+3).padStart(2,'0') + "-" + String(lastDay.getDate()).padStart(2,'0');
  } else if(range === 'fy') {
      var startY = m <= 2 ? y-1 : y;
      fDate = startY + "-04-01"; tDate = (startY+1) + "-03-31";
  }
  document.getElementById('gstrFrom').value = fDate; document.getElementById('gstrTo').value = tDate; generateGstReport();
}

function generateGstReport() {
  var tb = document.getElementById('reportTableBody'); var tf = document.getElementById('reportTableFoot'); if(!tb) return; tb.innerHTML = '';
  var fDate = document.getElementById('gstrFrom').value; var tDate = document.getElementById('gstrTo').value;
  var sumTax = 0, sumC = 0, sumS = 0, sumTot = 0;

  invoiceDB.forEach(inv => {
    if(inv.status !== 'Cancelled') {
      if(fDate && inv.dateIso < fDate) return; if(tDate && inv.dateIso > tDate) return;
      var tax = parseFloat(inv.taxableVal)||0;
      var tot = parseFloat(String(inv.grandTotal).replace(/[^0-9\.]/g, ''))||0;
      var cgst = tax * 0.09, sgst = tax * 0.09;
      sumTax += tax; sumC += cgst; sumS += sgst; sumTot += tot;

      tb.innerHTML += `<tr><td>${inv.invNo}</td><td>${inv.invDate}</td><td>${inv.custName}</td><td>${inv.custGst||'URP'}</td><td class="text-right">₹ ${tax.toFixed(2)}</td><td class="text-right">₹ ${cgst.toFixed(2)}</td><td class="text-right">₹ ${sgst.toFixed(2)}</td><td class="text-right bold">₹ ${tot.toFixed(2)}</td></tr>`;
    }
  });

  tf.innerHTML = `<tr><td colspan="4" class="text-right">TOTAL:</td><td class="text-right">₹ ${sumTax.toFixed(2)}</td><td class="text-right">₹ ${sumC.toFixed(2)}</td><td class="text-right">₹ ${sumS.toFixed(2)}</td><td class="text-right bold">₹ ${sumTot.toFixed(2)}</td></tr>`;
}

// --- CALCULATION & BILLING ENGINE ---
function calculateTotal() {
  var sum = 0, pcs = 0;
  document.querySelectorAll('#itemsBody tr.item-row').forEach((r, i) => {
    r.querySelector('.row-sl').innerText = i + 1;
    var q = parseFloat(r.querySelector('.item-qty').value)||0;
    var rate = parseFloat(r.querySelector('.item-rate').value)||0;
    var d = parseFloat(r.querySelector('.item-disc').value)||0;
    var amt = (q * rate) * (1 - d/100);
    r.querySelector('.item-amt').innerText = amt.toFixed(2);
    sum += amt; pcs += q;
  });
  document.getElementById('taxableAmount').innerText = sum.toFixed(2);
  var gst = sum * 0.18;
  document.getElementById('cgstAmount').innerText = (gst/2).toFixed(2);
  document.getElementById('sgstAmount').innerText = (gst/2).toFixed(2);
  var grand = Math.round(sum + gst);
  document.getElementById('roundOff').innerText = (grand - (sum + gst)).toFixed(2);
  document.getElementById('payableAmount').innerText = "₹ " + grand.toFixed(2);

  var upi = companyProfile.upi || "989773231@sbi";
  if(upi && grand > 0) {
    document.getElementById('upiQrCode').src = "https://api.qrserver.com/v1/create-qr-code/?size=100x100&margin=0&data=" + encodeURIComponent("upi://pay?pa=" + upi + "&pn=" + encodeURIComponent(companyProfile.name) + "&am=" + grand.toFixed(2));
    document.getElementById('upiQrCode').style.display = "inline-block";
    document.getElementById('qrPayAmt').innerText = grand.toFixed(2);
  }
}

function verifyTenantLogin() {
  var shopId = document.getElementById('loginShopIdInput').value.trim().toUpperCase();
  var pass = document.getElementById('loginPassInput').value.trim();
  var tenant = tenantAccounts.find(t => t.id === shopId && t.pass === pass);
  if(!tenant) return alert("❌ गलत क्लाइंट आईडी या पासवर्ड!");

  currentTenant = { id: tenant.id, name: tenant.name, perms: tenant.perms, role: (tenant.id === "MASTER" ? "superadmin" : "tenant") };
  localStorage.setItem('current_tenant_session', JSON.stringify(currentTenant));
  document.getElementById('passwordModal').style.display = 'none';
  location.reload();
}

  if(!url) return alert("URL डालें!");
  var m = url.match(/^.*(youtu.be\/|v\/|u\/\w\/|embed\/|watch\?v=|\&v=)([^#\&\?]*).*/);
  if(!m || m[2].length != 11) return alert("अमान्य YouTube लिंक!");

  document.getElementById('floatingVideoIframe').src = "https://www.youtube-nocookie.com/embed/" + m[2] + "?autoplay=1";

}

var voiceRec = null;
function toggleVoiceRecording() {
  var Speech = window.SpeechRecognition || window.webkitSpeechRecognition;
  if(!Speech) return alert("Chrome इस्तेमाल करें!");
  if(voiceRec) { voiceRec.stop(); voiceRec = null; document.getElementById('floatingMicBtn').classList.remove('listening'); return; }
  voiceRec = new Speech(); voiceRec.lang = 'hi-IN';
  document.getElementById('floatingMicBtn').classList.add('listening');
  document.getElementById('voiceReviewBar').style.display = 'flex';
  voiceRec.onresult = function(e) { document.getElementById('liveTranscriptInput').value = e.results[0][0].transcript; };
  voiceRec.start();
}
function pauseVoice() { if(voiceRec) { voiceRec.stop(); voiceRec = null; } document.getElementById('floatingMicBtn').classList.remove('listening'); }
function closeVoiceReview() { pauseVoice(); document.getElementById('voiceReviewBar').style.display = 'none'; }
function confirmAndExecuteCommand() {
  var cmd = document.getElementById('liveTranscriptInput').value.toLowerCase();
  closeVoiceReview();
  if(cmd.includes("बिल") || cmd.includes("print")) { switchSection('sale'); setTimeout(() => executePrint('bill'), 300); }
  else if(cmd.includes("स्टॉक") || cmd.includes("इन्वेंट्री")) switchSection('stock');
  else if(cmd.includes("खर्चा") || cmd.includes("expense")) switchSection('expense');
  else alert("कमांड मिली: " + cmd);
}
